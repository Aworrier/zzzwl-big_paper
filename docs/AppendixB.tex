\label{app:proof_appendix2}
%\addcontentsline{toc}{chapter}{忙期服务过程 ($Q \to P$) 的详细推导}

本附录分析节点忙期内缓存队列状态的演化过程，并构建马尔可夫吸收链模型\cite{2008_Markov_absorb}。该模型用于刻画任意节点的忙期开始时队列长度 $Q$ 与忙期结束时队列长度 $P$ 之间的演化关系，并进一步推导二者概率生成函数$G_P(z)$ 与 $G_Q(z)$ 的解析关系。

\section{构建马尔可夫吸收链模型}

为刻画忙期内队列变化及其终止机制，定义二维随机过程 $\{(L_t,K_t),t=0,1,\dots\}$。
\begin{itemize}
	\item $L_t \in \{0,1,\dots\}$ 表示忙期内节点第 $t$ 次发送数据包开始时刻的剩余队列长度，不计正在发送的数据包。若忙期初始总队列长度为 $Q$，则有 $L_0=Q-1$，这是由于队首数据包已占用信道。
	\item $K_t \in \{0,1\}$ 表示节点第 $t$ 次发送数据包时的信道状态。$K_t=1$ 表示忙期持续，即本次传输成功并进入下一轮发送；$K_t=0$ 表示忙期终止，即发生碰撞或发送结束。
\end{itemize}

如图 \ref{fig:absorbing_markov} 所示，该过程可建模为马尔可夫吸收链。
其状态空间划分如下：
\begin{itemize}
	\item \textbf{瞬态} $\mathcal{S}_T=\{(j,1)\mid j\ge 0\}$，表示节点持续成功发送且队列中仍有 $j$ 个待发包，系统在该状态下保持忙期。
	\item \textbf{吸收态} $\mathcal{S}_A=\{(j,0)\mid j\ge 0\}$，表示因碰撞或缓存队列清空而终止忙期。进入该状态后发送过程停止，此时的剩余队列长度即为 $P$。
\end{itemize}


\section[分析马尔可夫吸收链的状态转移概率]{分析马尔可夫吸收链的状态转移概率}

考虑系统处于瞬态 $(j,1)$。
此时节点正在发送一个数据包，缓存中还存在 $j$ 个数据包。
在一个时隙内，存在两类相互独立的过程：
\begin{enumerate}
	\item \textbf{新包到达过程}：新数据包以概率 $\lambda$ 到达，以概率 $1-\lambda$ 不到达。
	\item \textbf{传输过程}：其他 $n-1$ 个节点保持静默的概率为 $e^{-\theta}$，对应传输成功；发生碰撞的概率为 $1-e^{-\theta}$，对应传输失败。
\end{enumerate}

综合上述两个过程，分析可知下一时刻存在四种转移情形。

\begin{enumerate}
	\item \textbf{成功且有新到达}：概率为 $\lambda e^{-\theta}$。发送成功使队列长度减 1，新包到达使队列长度加 1，因此队列长度保持不变，状态由 $(j,1)$ 转移至 $(j,1)$。
	\item \textbf{成功且无新到达}：概率为 $(1-\lambda)e^{-\theta}$。发送成功且无补充使队列长度减 1，状态由 $(j,1)$ 转移至 $(j-1,1)$。
	\item \textbf{失败且有新到达}：概率为 $\lambda(1-e^{-\theta})$。碰撞使忙期立即结束，且新包已进入缓存，状态由 $(j,1)$ 转移至吸收态 $(j+1,0)$。
	\item \textbf{失败且无新到达}：概率为 $(1-\lambda)(1-e^{-\theta})$。碰撞结束忙期且无新包到达，状态由 $(j,1)$ 转移至吸收态 $(j,0)$。
\end{enumerate}
为后续推导方便，记 $b_{ij}$ 为从任意瞬态 $s_{t,i}$ 出发最终被吸收态 $s_{a,j}$ 吸收的概率。下一节将求解 $b_{ij}$ 的闭式表达式。

\section{求解吸收概率$b_{ij}$的闭式表达式}

为求取吸收概率，先将状态空间截断为有限维，设最大队列长度为 $J$，并在最后令 $J\to\infty$。转移矩阵 $\mathbf{M}$ 的标准分块形式为
\begin{equation}
	\mathbf{M} =
	\begin{pmatrix}
		\mathbf{T} & \mathbf{R} \\
		\mathbf{0} & \mathbf{I}
	\end{pmatrix}.
\end{equation}
其中：
\begin{itemize}
	\item $\mathbf{T}$ 为描述瞬态之间转移的子矩阵（下双对角矩阵），其元素为 $t_{j,j} = \lambda e^{-\theta}$、$t_{j, j-1} = (1-\lambda)e^{-\theta}$。
	\item $\mathbf{R}$ 为描述从瞬态进入吸收态的子矩阵，其元素为 $r_{j, j+1} = \lambda(1-e^{-\theta})$、$r_{j, j} = (1-\lambda)(1-e^{-\theta})$。
\end{itemize}

根据吸收马尔可夫链理论\cite{2008_Markov_absorb}，由概率 $b_{ij}$ 构成的吸收概率矩阵可表示为
$$
\mathbf{B}=(\mathbf{I}-\mathbf{T})^{-1}\mathbf{R}.
$$
利用 $\mathbf{T}$ 的下三角结构，可递归求解该矩阵方程。代入参数后，可得吸收概率 $b_{ij}$ 的闭式表达为：
\begin{itemize}
\item 当 $i=1$ 时，
\begin{equation}
b_{ij}=\left\{
\begin{array}{ll}
\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}, & j=0,\\
\dfrac{(1-\lambda)(1-e^{-\theta})}{1-\lambda e^{-\theta}}, & j=1,\\
\dfrac{\lambda(1-e^{-\theta})}{1-\lambda e^{-\theta}}, & j=2.
\end{array}
\right.
\end{equation}

\item 当 $i\ge 2$ 时，
\begin{equation}
b_{ij}=\left\{
\begin{array}{ll}
\left(\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}\right)^i, & j=0,\\[6pt]
\dfrac{(1-\lambda)(1-e^{-\theta})}{1-\lambda e^{-\theta}}\left(\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}\right)^{i-1}, & j=1,\\[6pt]
\dfrac{(1-\lambda)(1-e^{-\theta})}{(1-\lambda e^{-\theta})^2}\left(\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}\right)^{i-j}, & 2\le j\le i,\\[6pt]
\dfrac{\lambda(1-e^{-\theta})}{1-\lambda e^{-\theta}}, & j=i+1.
\end{array}
\right.
\end{equation}
\end{itemize}

\section[推导P对Q的条件概率分布以及条件PGF]{推导$P$对$Q$的条件概率分布以及条件PGF}
结合 $b_{ij}$ 可进一步分析 $Q$ 的演化。当 $Q=1$ 时，系统直接进入吸收态 $(0,0)$，即 $P=0$。当 $Q\ge 2$ 时，缓存队列的演化等价于从瞬态 $(Q-1,1)$ 出发。在考虑上述特殊情形后，可得到 $P$ 关于 $Q$ 的条件概率分布，具体为：当 $Q=1$ 时，$\Pr\{P=0 \mid Q=1\}=1$；当 $Q=2$ 时，
\begin{equation}
\Pr\{P=i \mid Q=2\}=\begin{cases}
\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}, & i=0,\\
\dfrac{(1-\lambda)(1-e^{-\theta})}{1-\lambda e^{-\theta}}, & i=1,\\
\dfrac{\lambda(1-e^{-\theta})}{1-\lambda e^{-\theta}}, & i=2.
\end{cases}
\label{eq:Pr_P_Q2_general}
\end{equation}
当 $Q=j\ge 3$ 时，
\begin{equation}
\Pr\{P=i \mid Q=j\}=\begin{cases}
\left[\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}\right]^{j-1}, & i=0,\\
\dfrac{(1-\lambda)(1-e^{-\theta})}{1-\lambda e^{-\theta}}\left[\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}\right]^{j-2}, & i=1,\\
\dfrac{(1-\lambda)(1-e^{-\theta})}{(1-\lambda e^{-\theta})^2}\left[\dfrac{(1-\lambda)e^{-\theta}}{1-\lambda e^{-\theta}}\right]^{j-i-1}, & 2\le i\le j-1,\\
\dfrac{\lambda(1-e^{-\theta})}{1-\lambda e^{-\theta}}, & i=j.
\end{cases}
\label{eq:Pr_P_Q3plus_general}
\end{equation}
当节点数量 $n$ 较大时，单个节点的输入率 $\lambda = \hat{\lambda}/n$ 趋近于零。此时，变量 $P$ 的概率分布可进一步近似表示为：
\begin{equation}
\operatorname{Pr}\{P=i\}=
\begin{cases}
\left(1-e^{-\theta}\right) e^{-\theta(Q-i-1)}, & i=1,\ldots, Q-1 \\
e^{-\theta(Q-1)}, & i=0.
\end{cases}
\end{equation}

进一步结合 PGF 的定义，可给出 $P$ 关于 $Q$ 的条件 PGF，见式 \eqref{eq:GQ_GP_relation}。
